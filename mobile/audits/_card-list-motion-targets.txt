  29 | // animatedPressableV1
  30 | // Reusable physical-button feedback without changing screen layout.
  31 | export function AnimatedPressable({
  32 |   children,
  33 |   style,
  34 |   pressedStyle,
  35 |   cue = 'tap',
  36 |   scaleTo = 0.975,
  37 |   onPress,
  38 |   disabled,
  39 |   ...props
  40 | }: AnimatedPressableProps) {
  41 |   const scale = useRef(
  42 |     new Animated.Value(1),
  43 |   ).current;
  44 | 
  45 |   const handlePressIn = () => {
  46 |     // Keep the shared motion helper as the default physical feel.

  48 |       pressIn(scale).start();
  49 |       return;
  50 |     }
  51 | 
  52 |     Animated.spring(scale, {
  53 |       toValue: scaleTo,
  54 |       damping: 18,
  55 |       stiffness: 320,
  56 |       mass: 0.45,
  57 |       useNativeDriver: true,
  58 |     }).start();
  59 |   };
  60 | 
  61 |   const handlePressOut = () => {
  62 |     pressOut(scale).start();
  63 |   };
  64 | 
  65 |   return (
  66 |     <Animated.View
  67 |       style={{
  68 |         transform: [{ scale }],
