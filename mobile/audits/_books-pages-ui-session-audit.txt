 398 | function dateWindow(dateText?: string) {
 399 |   const fallback = new Date();
 400 |   const parts = dateText?.match(/^(\d{4})-(\d{2})-(\d{2})$/);
 401 |   const start = parts
 402 |     ? new Date(Number(parts[1]), Number(parts[2]) - 1, Number(parts[3]))
 403 |     : new Date(fallback.getFullYear(), fallback.getMonth(), fallback.getDate());
 404 |   const end = new Date(start);
 405 |   end.setDate(end.getDate() + 1);
 406 |   const yyyy = start.getFullYear();
 407 |   const mm = String(start.getMonth() + 1).padStart(2, '0');
 408 |   const dd = String(start.getDate()).padStart(2, '0');
 409 |   return { start: start.getTime(), end: end.getTime(), date: `${yyyy}-${mm}-${dd}` };
 410 | }

 412 | function overlapSeconds(startedAt: string, endedAt: string | null, start: number, end: number) {
 413 |   const from = Math.max(Date.parse(startedAt), start);
 414 |   const to = Math.min(endedAt ? Date.parse(endedAt) : Date.now(), end);
 415 |   return Math.max(0, Math.floor((to - from) / 1000));
